title **Figure 1: Gauntlet Overview Diagram**\n<font size=12>A conceptual diagram of the MARL evaluation framework</font>

' Top row: Inputs
cloud "Policy Under Test (e.g., NN Module)" as PolicyInput
cloud "Evaluation\nEnvironments" as EnvInput

' Second row: Core Framework
package "Gauntlet Core Framework" {
  component "Configuration" as Config {
    "ContinualConfig"
    "EvaluationConfig"
  }
  component "EnhancedGauntletBenchmark" as Core {
    portin "evaluate_policy()"
  }
}

' Third row: Evaluation Engine
package "Evaluation Engine" {
  hexagon "Evaluation Loop" as EvalLoop
  "Episode Runner" as EpisodeRunner
  "Policy Under Test" as PUT
  "Environment" as Env
  "Challenger Agent" as Challenger
  "Metrics Tracker" as MetricsTracker
}

' Fourth row: Challenger Suite (horizontal layout)
package "Challenger Suite" {
  [Rule-Based Agents] as RuleBased
  [Neural Network Agents] as Neural  
  [Population-Based Agents] as Population
  [Adaptive Agents] as Adaptive
  
  note top of RuleBased
    - Fixed Strategy (e.g., AlwaysRock)
    - Pattern-Based (e.g., TitForTat)
  end note
  note top of Neural
   - Adversarial Networks
   - Specialist Exploiters
  end note
  note top of Population
   - Diverse Strategies
   - Evolutionary Selection
  end note
  note top of Adaptive
    - Counter-Exploiters
    - Meta-Learners
  end note
}

' Bottom row: Reporting
package "Reporting & Analysis" {
  component "Metrics Computation" as Metrics {
    "RobustnessMetrics"
  }
  database "Results History" as History
  component "Report Generator" as ReportGen {
    portout "generate_report()"
  }
  cloud "Comprehensive Report" as Output {
    artifact "Robustness Radar Chart" as Radar
    artifact "Performance Heatmap" as Heatmap
    artifact "Weakness Analysis" as Weakness
    artifact "Improvement Suggestions" as Suggestions
  }
}

' Main flow connections (top to bottom)
PolicyInput ..> Core : <font size=10>Submits</font>
Core -down-> EvalLoop : <font size=10>Initiates</font>
Config -right-> Core : <font size=10>Configures</font>
Core -right-> EnvInput : <font size=10>Uses</font>

' Evaluation loop internal connections
EvalLoop -down-> EpisodeRunner : <font size=10>Runs N episodes</font>
EpisodeRunner -right-> PUT : <font size=10>act(state)</font>
PUT -down-> Env : <font size=10>action</font>
Env -left-> PUT : <font size=10>next_state, reward</font>
EpisodeRunner -down-> Challenger : <font size=10>act(state, history)</font>
Challenger -up-> Env : <font size=10>action</font>
Env -down-> Challenger : <font size=10>next_state, reward</font>
EpisodeRunner ..> MetricsTracker : <font size=10>Logs raw data</font>

' Challenger connections (from evaluation loop to challengers)
EvalLoop -down-> RuleBased : <font size=10>Selects</font>
EvalLoop -down-> Neural : <font size=10>Selects</font>
EvalLoop -down-> Population : <font size=10>Selects</font>
EvalLoop -down-> Adaptive : <font size=10>Selects</font>

' Reporting connections
Core -down-> Metrics : <font size=10>Computes</font>
Metrics -right-> History : <font size=10>Stores</font>
Metrics -down-> ReportGen : <font size=10>Provides data for</font>
ReportGen -left-> History : <font size=10>Reads</font>
ReportGen ..> Output : <font size=10>Generates</font>

' Layout hints to reduce intersections
PolicyInput -[hidden]-> Config
Config -[hidden]-> EnvInput
RuleBased -[hidden]right-> Neural
Neural -[hidden]right-> Population
Population -[hidden]right-> Adaptive
Metrics -[hidden]-> ReportGen
